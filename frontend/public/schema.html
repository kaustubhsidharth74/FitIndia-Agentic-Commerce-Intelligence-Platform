SQL — Quick Queries

-- all audit entries with customer name
SELECT a.*, c.name AS customer_name
FROM audit_log a
LEFT JOIN customers c ON a.customer_id = c.id
ORDER BY a.timestamp DESC LIMIT 100;

-- revenue recovered by agents
SELECT COALESCE(SUM(amount_paise), 0) / 100.0 AS recovered_inr
FROM audit_log
WHERE result = 'success'
AND action_type IN ('upsell', 'campaign_converted', 'payment_retry');

-- paid orders with product name
SELECT o.*, p.name AS product_name, o.total_paise / 100.0 AS total_inr
FROM orders o
LEFT JOIN products p ON o.product_id = p.id
WHERE o.status = 'paid' ORDER BY o.created_at DESC;
